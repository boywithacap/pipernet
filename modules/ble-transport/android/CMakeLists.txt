cmake_minimum_required(VERSION 3.22.1)
project(piperble C)
set(SHARED "${CMAKE_CURRENT_SOURCE_DIR}/../shared")
file(READ "${SHARED}/sources.json" SOURCE_LIST)
string(JSON COUNT LENGTH "${SOURCE_LIST}")
math(EXPR LAST "${COUNT} - 1")
set(SOURCES "piperble-jni.c")
foreach(INDEX RANGE ${LAST})
  string(JSON SOURCE GET "${SOURCE_LIST}" ${INDEX})
  list(APPEND SOURCES "${SHARED}/${SOURCE}")
endforeach()
add_library(piperble SHARED ${SOURCES})
target_compile_features(piperble PRIVATE c_std_11)
target_compile_definitions(piperble PRIVATE ED25519_CUSTOMHASH ED25519_CUSTOMRANDOM)
target_include_directories(piperble PRIVATE "${SHARED}" "${SHARED}/vendor/noise/include" "${SHARED}/vendor/noise/src" "${SHARED}/vendor/noise/src/protocol")
# Required by Android devices using 16 KiB memory pages.
target_link_options(piperble PRIVATE "-Wl,-z,max-page-size=16384")
